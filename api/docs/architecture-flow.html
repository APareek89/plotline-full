  UI["Campaign workspace [FUNCTION]<br/>in: conversation or explicit action<br/>out: same-origin request"]
  LOGIN["Credentials sign-in [LIBRARY · Auth.js]<br/>in: password plus auth CSRF<br/>out: HttpOnly session cookie"]
  SESSION["Revocable session [DATA · PostgreSQL]<br/>in: session identifier<br/>out: current owner or denial"]
  BFF{"Actor gate [FUNCTION]<br/>in: cookie, Origin, CSRF<br/>out: allow only current session and matching mutation origin"}
  PROOF["Request proof [FUNCTION]<br/>in: owner, session, method, path, body hash<br/>out: short-lived signed actor proof"]
  API{"Private API gate [FUNCTION]<br/>in: signed proof plus actual request<br/>out: allow only matching proof, nonce and active session"}
  SCOPE["Owner transaction [LIBRARY · PostgreSQL]<br/>in: verified actor<br/>out: transaction-local owner scope"]
  ROWS["Owner policies [DATA · PostgreSQL RLS]<br/>in: owner-scoped query<br/>out: only that owner's state"]
  FLOW["Campaign workflow [FUNCTION]<br/>in: current state plus event<br/>out: saved artifacts and next question<br/>see diagram10"]
  FILE["Private object references [DATA · S3]<br/>in: owner, version and SHA256<br/>out: immutable source or generated bytes"]
  MEDIA["Authorized asset response [FUNCTION]<br/>in: owned asset reference<br/>out: same-origin private file or bundle"]
  CLIENT["Response fence [FUNCTION]<br/>in: body plus original account generation<br/>out: current-account UI only"]
  STOP["Safe denial [FUNCTION]<br/>in: invalid or revoked boundary<br/>out: no provider dispatch"]

  UI --> BFF
  SESSION --> BFF

  API -->|"valid"| SCOPE --> ROWS --> FLOW
  API -->|"invalid"| STOP
  FLOW --> ROWS
  ROWS --> MEDIA
  FILE --> MEDIA --> CLIENT
  FLOW --> CLIENT
  CLIENT -.->|"next explicit reply"| UI
